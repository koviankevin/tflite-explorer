import assert from 'node:assert/strict';
import {createHash} from 'node:crypto';
import {runtimeLoopFixture} from '../../test/runtime-loop-fixture.mjs';
export async function assertActiveCompute30(r){
    assert.equal(r.activeComputeRequired,true);const a=r.activeCompute;assert.equal(a.error,undefined);assert.equal(a.deadlineMs,30000);assert.equal(a.networkHeld,false);assert.equal(a.fakeTimers,false);assert.equal(a.visibility,'visible');assert.equal(a.mainThreadResponsive,true);assert.ok(a.uiResponseMs>=0&&a.uiResponseMs<5000);
    assert.deepEqual(a.cases.map(c=>c.kind),['finite-control','in-kernel-cancel','in-kernel-timeout','finite-recovery']);assert.deepEqual(a.cases.map(c=>c.status),['passed','not-verified','not-verified','passed']);assert.equal(a.modelWorkers.length,3);assert.equal(a.runtimeWorkers.length,4);assert.ok([...a.modelWorkers,...a.runtimeWorkers].every(w=>w.closed&&w.closedAt>=w.startedAt&&w.lifetimeMs===w.closedAt-w.startedAt));const workerBase='https://koviankevin.github.io/tflite-explorer/versions/'+r.expectedBuild+'/src/';assert.ok(a.modelWorkers.every(w=>w.url===workerBase+'model-worker.js?v='+r.expectedBuild));assert.ok(a.runtimeWorkers.every(w=>w.url===workerBase+'tflite-runtime-worker.js?v='+r.expectedBuild));
    const fixtures=await Promise.all([runtimeLoopFixture(false),runtimeLoopFixture(true)]),hash=b=>createHash('sha256').update(b).digest('hex');assert.equal(a.fixture.bytes,fixtures[0].bytes.length);assert.equal(a.fixture.finiteSHA256,hash(fixtures[0].bytes));assert.equal(a.fixture.infiniteSHA256,hash(fixtures[1].bytes));const delta=[];for(let i=0;i<fixtures[0].bytes.length;i++)if(fixtures[0].bytes[i]!==fixtures[1].bytes[i])delta.push(i);assert.deepEqual(delta,[a.fixture.onlyChangedByte]);assert.equal(a.fixture.falseByte,0);assert.equal(a.fixture.trueByte,1);assert.equal(a.fixture.subgraphs,3);assert.deepEqual(a.fixture.builtins,['WHILE','ADD']);assert.deepEqual(a.fixture.input,{dtype:'float32',shape:[1]});
    for(const i of [0,3]){const io=a.cases[i].io;assert.deepEqual(io.output,[7]);assert.equal(io.modelSHA256,a.fixture.finiteSHA256);assert.match(io.sha256,/^[a-f0-9]{64}$/);}
    const cancel=a.cases[1],timeout=a.cases[2];assert.equal(cancel.invokeObserved,true);assert.equal(cancel.mainThreadResponsive,true);assert.equal(cancel.downloadDisabled,true);assert.ok(cancel.cancelToCloseMs>=0&&cancel.cancelToCloseMs<5000);assert.equal(timeout.invokeObserved,true);assert.equal(timeout.downloadDisabled,true);assert.match(timeout.reason,/超過 30 秒.*終止 Worker/);assert.ok(timeout.workerLifetimeMs>=25000&&timeout.workerLifetimeMs<=40000);assert.equal(timeout.workerLifetimeMs,a.runtimeWorkers[2].lifetimeMs);
    for(const [i,stamp]of [[1,a.cancelInvokeObservedAt],[2,a.timeoutInvokeObservedAt]])assert.ok(Number.isFinite(stamp)&&stamp>=a.runtimeWorkers[i].startedAt&&stamp<a.runtimeWorkers[i].closedAt);
    assert.equal(a.screenshot,true);return {cases:4,fixtureBytes:a.fixture.bytes,finiteSHA256:a.fixture.finiteSHA256,infiniteSHA256:a.fixture.infiniteSHA256,invokeObserved:true,mainThreadResponsive:true,uiResponseMs:a.uiResponseMs,cancelToCloseMs:cancel.cancelToCloseMs,timeoutWorkerLifetimeMs:timeout.workerLifetimeMs,deadlineMs:30000,recoveryOutput:[7],modelWorkersClosed:3,runtimeWorkersClosed:4,networkHeld:false,fakeTimers:false};
}
