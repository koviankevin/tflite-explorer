function Write-Durable([string]$taskPath,[string]$taskText,[bool]$taskAppend=$false){

    $taskStream=[IO.FileStream]::new($taskPath,$taskMode,[IO.FileAccess]::Write,[IO.FileShare]::Read)
    try{$taskBytes=$taskEncoding.GetBytes($taskText);$taskStream.Write($taskBytes);$taskStream.Flush($true)}finally{$taskStream.Dispose()}
}

    if($taskInfo.Attributes -band [IO.FileAttributes]::ReparsePoint){
        $taskTagLines=@(fsutil reparsepoint query $taskFull);$taskTagLine=$taskTagLines[0]
        if($LASTEXITCODE -ne 0 -or $taskTagLine -notmatch '0x([0-9a-fA-F]+)'){throw 'Reparse tag unreadable'}

}
