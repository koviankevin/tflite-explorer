import assert from 'node:assert/strict';
import {createHash} from 'node:crypto';
import {runtimeMemoryFixture,runtimeMemoryValues,RUNTIME_MEMORY_ELEMENTS} from '../../test/runtime-memory-fixture.mjs';
import {evaluateRuntimeLifecycle,RUNTIME_LIFECYCLE_POLICY as policy} from '../../test/browser/runtime-lifecycle-policy.mjs';
export async function assertRuntimeIO30(raw){
    assert.equal(raw.runtimeLifecycleRequired,true);const r=raw.runtimeLifecycle;assert.equal(r.error,undefined);const evaluated=evaluateRuntimeLifecycle(r);assert.equal(evaluated.status,'passed',JSON.stringify(evaluated));assert.deepEqual(r.evaluation,evaluated);
    const hash=b=>createHash('sha256').update(b).digest('hex'),f=await runtimeMemoryFixture();assert.deepEqual(r.fixture,{modelBytes:f.bytes.length,modelSHA256:hash(f.bytes),weightBytes:4194304,inputBytes:4194304,outputBytes:4194304,shape:[1,1024,1024],dtype:'float32',operator:'ADD'});
    for(const c of r.cycles){const input=c.cycle*3-1;assert.equal(c.io.inputSHA256,hash(new Float32Array(RUNTIME_MEMORY_ELEMENTS).fill(input)));assert.equal(c.io.outputSHA256,hash(runtimeMemoryValues(input)));}
    const base='https://koviankevin.github.io/tflite-explorer/versions/'+raw.expectedBuild+'/src/';assert.ok(r.modelWorkers.every(w=>w.url===base+'model-worker.js?v='+raw.expectedBuild));assert.ok(r.runtimeWorkers.every(w=>w.url===base+'tflite-runtime-worker.js?v='+raw.expectedBuild));
    assert.equal(r.urls.length,8);assert.ok(r.urls.every(u=>u.bytes>=8388608&&u.bytes<8404992&&u.createdAt>=Date.parse(raw.startedAt)&&u.revokedAt<=Date.parse(raw.finishedAt)));
    const p=raw.processCleanup;assert.equal(p.allReaped,true);assert.deepEqual(p.remaining,[]);assert.ok(p.observations>0);assert.ok(p.identities.length>=2&&p.identities.length<=1024);assert.deepEqual(p.identities,r.ownedProcesses);assert.equal(raw.processOwnership.format,'model-lens-owned-processes');assert.equal(raw.processOwnership.version,1);assert.ok(p.identities.some(i=>i.pid===raw.processOwnership.root.pid&&i.startTicks===raw.processOwnership.root.startTicks));const keys=p.identities.map(i=>{assert.ok(Number.isSafeInteger(i.pid)&&i.pid>0);assert.match(i.startTicks,/^\d+$/);return i.pid+':'+i.startTicks;});assert.equal(new Set(keys).size,keys.length);
    assert.ok(r.samples.every(s=>Number.isFinite(s.mainHeapBytes)&&Number.isFinite(s.rssBytes)&&s.processes<=p.identities.length));assert.equal(r.policy.deadlineMs,270000);assert.deepEqual(r.policy,policy);
    return {cycles:8,warmups:2,manualCPU:24,fullIOByteComparisons:8,modelBytes:r.fixture.modelBytes,modelSHA256:r.fixture.modelSHA256,inputBytes:4194304,outputBytes:4194304,elapsedMs:r.elapsedMs,summary:evaluated.summary,samples:r.samples.length,modelWorkersClosed:8,runtimeWorkersClosed:24,nativeURLLifetimeMs:r.urls.map(u=>u.revokedAt-u.createdAt),observedProcessIdentities:p.identities.length,allObservedReaped:true,forcedGC:false,shortenedURLTimer:false,scope:'Hosted Linux Chromium finite sample. Summed owned process RSS may double-count shared pages; main V8 heap is not continuous runtime Worker/WASM memory.'};
}
